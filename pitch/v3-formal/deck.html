theorem verdict_accept_iff … : verdict cfg e = .accept ↔ AcceptSpec cfg e

theorem verdict_reject_sound … (h : verdict cfg e = .reject) : RejectSpec cfg e

→ every does not meet rests on a documented violation

theorem fail_needs_trusted … (h : checkCapability k c = .fail) : TrustedEvidence c.evidence

→ every failure rests on trusted evidence

theorem verdict_refines (hr : Refines cfg cfg') (hd : verdict cfg e ≠ .needMoreInfo) : verdict cfg' e = verdict cfg e

→ new data never flips a decided verdict
